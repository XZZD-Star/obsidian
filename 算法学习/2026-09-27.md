#include<iostream>
#include<vector>

using namespace std;

const int MAXN = 1000005;

struct Order
{
    long long d;
    int s;
    int t;
} orders[MAXN];

int r[MAXN];
int diff[MAXN];
int n,m;
int ans;

bool check(int k)
{
    fill(diff,diff+n+2,0);
    
    for(int i = 1;i <= k;++i)
    {
        diff[orders[i].s] += orders[i].d;
        diff[orders[i].t+1] -= orders[i].d;
    }
    
    long long current = 0;
    for(int i = 1;i <= n;++i)
    {
        current += diff[i];
        if(current > r[i])
        {
            return false;
        }
    }
    return true;
}
int main()
{
    // 优化输入输出速度
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    if(!(cin >> n >> m)) return 0;
    
    for(int i = 1;i <= n;++i)
    {
        cin >> r[i];
    }
    
    for(int i = 1;i <= m; ++i)
    {
        cin >> orders[i].d >> orders[i].s >> orders[i].t;
    }
    
    int left = 1;
    int right = m;
    while(left <= right)
    {
        int mid = left + (right - left) / 2;
        if(check(mid))
        {
            left += 1;
        }
        else
        {
            ans = mid;
            right = mid - 1;
        }
    }
    
    cout << ans << endl;
    return 0;
}
