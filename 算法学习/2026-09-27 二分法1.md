
## 入门班借教室

关键：
- 差分就是取变化点
- 理清二分到底二分的内容是什么是天数还是订单
- 掌握模板两类情况求第一个满足或最后一个满足；同时需要判断端点边界情况，比如最后一个不满足，如果第一个也不满足呢？！
- 理解current表示是当前天数的实际需求
  
```C++
#include<iostream>
#include<algorithm>
using namespace std;

const int MAXN = 1000005;

int n, m;

int r[MAXN];
long long diff[MAXN];

struct Order
{
    int d;
    int s;
    int t;
} orders[MAXN];

bool check(int k)
{
    fill(diff, diff + n + 2, 0);

    for(int i = 1; i <= k; ++i)
    {
        diff[orders[i].s] += orders[i].d;
        diff[orders[i].t + 1] -= orders[i].d;
    }

    long long current = 0;

    for(int i = 1; i <= n; ++i)
    {
        current += diff[i];

        if(current > r[i])
            return false;
    }

    return true;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;

    for(int i = 1; i <= n; ++i)
    {
        cin >> r[i];
    }

    for(int i = 1; i <= m; ++i)
    {
        cin >> orders[i].d
            >> orders[i].s
            >> orders[i].t;
    }
    // 找最后一个能够满足的订单
    int s = 0;
    int e = m;

    while(s < e)
    {
        int mid = s + e + 1 >> 1;

        if(check(mid))
        {
            s = mid;
        }
        else
        {
            e = mid - 1;
        }
    }

    // 所有订单都可以满足
    if(s == m)
        cout << 0;
    else
        cout << s + 1;

    return 0;
}
```

## 一元三次方程求解NC16694



```
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

double a, b, c, d;

double f(double x)
{
    return a * x * x * x + b * x * x + c * x + d;
}

int main()
{
    cin >> a >> b >> c >> d;

    for(int i = -100; i <= 100; ++i)    
    {
        double l = i;
        double r = i + 1;

        // 左端点就是根
        if(fabs(f(l)) == 0)
        {
            printf("%.2f ", l);
            continue;
        }

        // 100 已经是最后一个点，没有 [100,101]
        if(i == 100)
            break;

        // 只有严格异号，根才在区间内部
        if(f(l) * f(r) >= 0)
            continue;

        // 浮点数二分
        for(int j = 0; j < 100; ++j)
        {
            double mid = (l + r) / 2;

            if(f(l) * f(mid) <= 0)
                r = mid;
            else
                l = mid;
        }

        printf("%.2f ", l);
    }

    return 0;
}
```
